import { readFileSync, writeFileSync } from 'node:fs';
const U = 'https://m0-15-spike.neurocapacitar-projetos.workers.dev';
const key = readFileSync('.spike-key', 'utf8');
const sleep = (ms) => new Promise((r) => setTimeout(r, ms));
async function gen(variant, body = {}, save) {
  const t = Date.now();
  const r = await fetch(`${U}/pdf/${variant}`, { method: 'POST', headers: { 'x-spike-key': key, 'content-type': 'application/json' }, body: JSON.stringify(body) });
  const wall = Date.now() - t;
  const buf = Buffer.from(await r.arrayBuffer());
  if (save && r.status === 200) writeFileSync(save, buf);
  const meta = r.status === 200 ? { bytes: buf.length, sha: r.headers.get('x-sha256')?.slice(0, 12), serverMs: r.headers.get('x-ms') } : buf.toString('utf8').slice(0, 300);
  console.log(variant.padEnd(18), r.status, `wall=${wall}ms`, JSON.stringify(meta));
  return { status: r.status, wall, sha: r.headers.get('x-sha256') };
}
const mode = process.argv[2];
if (mode === 'basic') {
  await gen('ok', {}, 'cert2.pdf'); await sleep(11000);
  await gen('ok', { name: '<script>document.title="X"</script><img src=https://example.com/p.png>Maria "Silva" & Cia' }, 'xss-escaped.pdf'); await sleep(11000);
  await gen('unescaped', {}, 'xss-unescaped.pdf'); await sleep(11000);
  await gen('external-allowed', {}, 'external-allowed.pdf'); await sleep(11000);
  await gen('a4-portrait', {}, 'portrait.pdf'); await sleep(11000);
  await gen('timeout'); await sleep(11000);
  await gen('huge');
} else if (mode === 'extra') {
  await gen('unescaped', {}, 'csp-unescaped.pdf'); await sleep(11000);
  await gen('ok', {}, 'csp-ok.pdf'); await sleep(11000);
  await gen('badformat');
} else if (mode === 'repeat') {
  const shas = new Set(); const walls = [];
  for (let i = 0; i < 6; i++) { const x = await gen('ok', {}, i === 0 ? 'rep.pdf' : undefined); shas.add(x.sha); walls.push(x.wall); await sleep(11000); }
  console.log('SHAs distintos nas 6 repetições (mesmo HTML):', shas.size, '| wall ms:', walls.join(' '));
} else if (mode === 'concurrent') {
  await Promise.all([gen('ok'), gen('ok'), gen('ok')]);
}
